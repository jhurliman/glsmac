CMAKE_MINIMUM_REQUIRED( VERSION 3.13 )
PROJECT( GLSMAC_movie_probe LANGUAGES CXX )
IF ( NOT UNIX OR APPLE )
	MESSAGE( FATAL_ERROR "This focused packaging/cancellation probe requires Linux" )
ENDIF ()
GET_FILENAME_COMPONENT( GLSMAC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE )
ADD_EXECUTABLE( ${PROJECT_NAME} "${GLSMAC_ROOT}/tools/tests/movie_playback_probe.cpp" "${GLSMAC_ROOT}/src/task/intro/Movie.cpp" )
TARGET_COMPILE_FEATURES( ${PROJECT_NAME} PRIVATE cxx_std_17 )
TARGET_INCLUDE_DIRECTORIES( ${PROJECT_NAME} PRIVATE "${GLSMAC_ROOT}/src" )
FIND_PACKAGE( PkgConfig REQUIRED )
PKG_CHECK_MODULES( SDL REQUIRED IMPORTED_TARGET sdl2 )
FIND_PACKAGE( Threads REQUIRED )
TARGET_LINK_LIBRARIES( ${PROJECT_NAME} PRIVATE PkgConfig::SDL Threads::Threads )
INCLUDE( "${GLSMAC_ROOT}/cmake/Movies.cmake" )
ENABLE_TESTING()
ADD_TEST( NAME movie_probe_missing COMMAND ${PROJECT_NAME} /__glsmac_missing_movie__/opening.wve )
SET_TESTS_PROPERTIES( movie_probe_missing PROPERTIES WILL_FAIL TRUE TIMEOUT 5 ENVIRONMENT "SDL_AUDIODRIVER=dummy" )
